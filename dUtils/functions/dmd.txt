function showIntro():Void{_root.gotoAndStop("intro");}
function startGame():Void{_root.gotoAndStop("game");}
function playMusic():Void{mySound.start();}
function stopMusic():Void{mySound.stop();}                 //someone please test these... for outputs..

Sequencer.Start(this, function() {
	Sequencer.Call(showIntro);
	Sequencer.Wait(4000);
	Sequencer.Call(stopMusic);
	Sequencer.Wait(500);
	Sequencer.Call(playMusic);
	Sequencer.Call(startGame);
});

//5

function setAlpha(clip:MovieClip, value:Number):Void {
	clip._alpha = value;
}

Sequencer.Start(this, function() {
	Sequencer.Call(setAlpha, [player_mc, 0]);
	Sequencer.Wait(500);
	Sequencer.Call(setAlpha, [player_mc, 100]);
});
